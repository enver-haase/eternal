# gen_init_cpio device table for the MMU BusyBox initramfs.
# Static ET_EXEC busybox at page-1 — no shared libs needed.
dir /bin 0755 0 0
dir /sbin 0755 0 0
dir /etc 0755 0 0
dir /proc 0755 0 0
dir /sys 0755 0 0
dir /dev 0755 0 0
dir /root 0700 0 0
dir /tmp 1777 0 0

# Device nodes (console is required for init's stdin/stdout/stderr)
nod /dev/console 0600 0 0 c 5 1
nod /dev/null 0666 0 0 c 1 3
nod /dev/zero 0666 0 0 c 1 5
nod /dev/ttyS0 0666 0 0 c 4 64
nod /dev/tty 0666 0 0 c 5 0
nod /dev/tty0 0620 0 0 c 4 0
nod /dev/fb0 0660 0 0 c 29 0
# (/dev/dsp + /dev/opl are provided by devtmpfs, which /doom mounts at startup.)

# The static busybox and init (busybox dispatches on argv[0] basename)
file /bin/busybox ../busybox/busybox 0755 0 0
# /init forks the game and execs a shell when it exits, so quitting DOOM leaves you at a
# prompt on a machine that is still up (init/mmuinit.c explains why neither a #! script nor
# BusyBox init works here).
file /init ../init/mmuinit 0755 0 0
slink /sbin/init /bin/busybox 0777 0 0
slink /bin/sh /bin/busybox 0777 0 0

# Sound-device self-test (static ET_EXEC) — kept as a diagnostic tool.
file /sndtest ../sndtest 0755 0 0

# fbdoom (static ET_EXEC, page-1). The game finds its IWAD in $DOOMWADDIR, so each edition
# lives in its own directory with exactly its own WAD -- the file NAME is what tells the game
# which edition it is (doomu.wad = Ultimate/retail, doom.wad = registered, doom1.wad =
# shareware, doom2.wad = Doom II), so one per directory keeps them from shadowing each other.
#
# /wadrun dispatches on argv[0] and is linked as one command per edition. The licensed WADs are
# NOT in the repository -- they live in wads/ on the build host, which .gitignore excludes.
file /doom ../doom/doom 0755 0 0
file /wadrun ../wadrun 0755 0 0
dir /wads 0755 0 0
dir /wads/shareware 0755 0 0
dir /wads/registered 0755 0 0
dir /wads/ultimate 0755 0 0
dir /wads/doom2 0755 0 0
file /wads/shareware/doom1.wad ../wads/shareware/doom1.wad 0644 0 0
file /wads/registered/doom.wad ../wads/registered/doom.wad 0644 0 0
file /wads/ultimate/doomu.wad ../wads/ultimate/doomu.wad 0644 0 0
file /wads/doom2/doom2.wad ../wads/doom2/doom2.wad 0644 0 0
slink /bin/doom /wadrun 0777 0 0
slink /bin/doom19 /wadrun 0777 0 0
slink /bin/udoom /wadrun 0777 0 0
slink /bin/doom2 /wadrun 0777 0 0
