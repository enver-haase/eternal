# Device nodes (console is required for init's stdin/stdout/stderr)
nod /dev/console 0600 0 0 c 5 1
nod /dev/null 0666 0 0 c 1 3
nod /dev/zero 0666 0 0 c 1 5
nod /dev/ttyS0 0666 0 0 c 4 64
nod /dev/tty 0666 0 0 c 5 0
nod /dev/tty0 0620 0 0 c 4 0
# Virtual terminals 1-4. SDL 1.2 asks the VT layer for a free VT (VT_OPENQRY) and then opens
# /dev/ttyN -- with only tty0 present that open fails and SDL reports "Unable to open a console
# terminal", which reads like a driver problem and is a missing device node.
nod /dev/tty1 0620 0 0 c 4 1
nod /dev/tty2 0620 0 0 c 4 2
nod /dev/tty3 0620 0 0 c 4 3
nod /dev/tty4 0620 0 0 c 4 4
nod /dev/fb0 0660 0 0 c 29 0
# (/dev/dsp + /dev/opl are provided by devtmpfs, which /doom mounts at startup.)
